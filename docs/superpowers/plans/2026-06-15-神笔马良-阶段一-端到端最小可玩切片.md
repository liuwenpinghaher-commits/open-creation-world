# 神笔马良 · 阶段一（端到端最小可玩切片）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让"孩子画简笔画 → 语音说 → 多模态 LLM（默认豆包）理解 → AI 用同风格简笔画回应（新增一个元素 + 让它动一下）+ 语音讲一句故事"这条端到端链路真正跑通，成为可玩的第一版骨架。

**Architecture:** Vite + React + TypeScript 前端（tldraw 画布 + 在线语音）+ 一个轻量 Node 后端代理（统一封装可插拔多模态模型与在线语音、保护 key）。前端把"画布快照图 + 孩子语音转写 + 历史"发给后端，后端调多模态模型返回结构化 `CocreateResponse`，前端把它增量应用到画布（新增/修改简笔画、跑动画）并用 TTS 讲出来。纯逻辑（状态归约、简笔画指令映射）走 TDD；外部 IO（模型、语音、tldraw 渲染）走实现 + 手动端到端验证。

**Tech Stack:** Vite, React 19, TypeScript, tldraw, rough.js, Dexie（后续阶段用）, Express, Vitest, 火山方舟豆包 `doubao-seed-2-pro`（OpenAI 兼容接口）, 在线语音（STT/TTS，可插拔）。

**配套文档:** 产品 `PRD-神笔马良-简笔画想象力共创.md`(v1.1)；设计 `docs/superpowers/specs/2026-06-15-创造世界-评审与优化方案-design.md`(v3.1)。

/** 一个简笔画图形原语（限定有限集合，保证 AI 产出可控、可手绘化） */
export type SketchShapeSpec = {
  id: string
  kind: 'draw' | 'ellipse' | 'rectangle' | 'line' | 'arrow' | 'text'
  /** 归一化坐标，0..1，渲染时映射到画布尺寸 */
  x: number
  y: number
  w?: number
  h?: number
  points?: { x: number; y: number }[] // draw/line 用
  text?: string
  color?: string
  label?: string
}

export type ShapePatch = {
  id: string
  remove?: boolean
  x?: number
  y?: number
  color?: string
  text?: string
}

export type AnimationStep = {
  shapeId: string
  at: number // 第几拍，从 0 起
  x?: number
  y?: number
  rotation?: number
  opacity?: number
  durationMs?: number
}

export type GuidanceDimension =
  | 'observe_express'
  | 'narrative_logic'
  | 'imagination'
  | 'problem_solving'

export type CocreateResponse = {
  understanding: string
  guidanceIntent?: { dimension: GuidanceDimension; move: string }
  askChild?: string
  addShapes?: SketchShapeSpec[]
  modifyShapes?: ShapePatch[]
  animation?: AnimationStep[]
  story?: { append?: string; rewrite?: { from: number; text: string } }
  speak: string
}

export type StoryEntry = { id: string; text: string }

export type SessionState = {
  shapes: SketchShapeSpec[]
  story: StoryEntry[]
  dialog: { role: 'child' | 'system'; text: string }[]
}

// src/store/sessionStore.ts
import type { CocreateResponse, SessionState, SketchShapeSpec } from '../types/cocreate'

export function emptySession(): SessionState {
  return { shapes: [], story: [], dialog: [] }
}

let storyCounter = 0
function nextStoryId(): string {
  storyCounter += 1
  return `story:${storyCounter}`
}

export function applyResponse(
  state: SessionState,
  res: CocreateResponse,
  childUtterance: string
): SessionState {
  let shapes: SketchShapeSpec[] = [...state.shapes]

  if (res.addShapes) shapes.push(...res.addShapes)

  if (res.modifyShapes) {
    for (const patch of res.modifyShapes) {
      if (patch.remove) {
        shapes = shapes.filter((s) => s.id !== patch.id)
        continue
      }
      shapes = shapes.map((s) =>
        s.id === patch.id
          ? { ...s, ...('x' in patch ? { x: patch.x! } : {}), ...('y' in patch ? { y: patch.y! } : {}), ...(patch.color ? { color: patch.color } : {}), ...(patch.text ? { text: patch.text } : {}) }
          : s
      )
    }
  }

  let story = [...state.story]
  if (res.story?.append) story.push({ id: nextStoryId(), text: res.story.append })
  if (res.story?.rewrite) {
    const { from, text } = res.story.rewrite
    story = story.slice(0, from).concat({ id: nextStoryId(), text })
  }

  const dialog = [...state.dialog]
  if (childUtterance) dialog.push({ role: 'child', text: childUtterance })
  if (res.speak) dialog.push({ role: 'system', text: res.speak })

  return { shapes, story, dialog }
}
```

// server/model.ts
import type { CocreateResponse } from '../src/types/cocreate'

const SYSTEM_PROMPT = `你是"神笔马良"里的共创伙伴，陪 6-10 岁孩子用简笔画编故事。
规则：
- 你画的也是简笔画，只用有限图形原语：draw/ellipse/rectangle/line/arrow/text。坐标用 0..1 归一化。
- 看孩子的画 + 听他说的话，理解他想做什么；不清楚就用 askChild 问一句（口语、亲切）。
- 让画面动起来：用 animation 给图形安排简单移动/旋转/淡入。
- 和孩子一起编故事：story.append 追加一小段；speak 是你要说出来的话（口语、简短、鼓励）。
- 成长引导（均衡、引导而非裁决、孩子可忽略）：在合适时机用 guidanceIntent 选一个维度
  (observe_express/narrative_logic/imagination/problem_solving) 并把引导自然融进 speak。
- 只输出 JSON，符合给定结构，不要多余文字。`

export async function callModel(input: {
  imageDataUrl: string
  childUtterance: string
  history: { role: 'child' | 'system'; text: string }[]
}): Promise<CocreateResponse> {
  const body = {
    model: process.env.MODEL_ID,
    messages: [
      { role: 'system', content: SYSTEM_PROMPT },
      ...input.history.map((h) => ({
        role: h.role === 'child' ? 'user' : 'assistant',
        content: h.text,
      })),
      {
        role: 'user',
        content: [
          { type: 'text', text: input.childUtterance || '（孩子在画画）' },
          { type: 'image_url', image_url: { url: input.imageDataUrl } },
        ],
      },
    ],
    response_format: { type: 'json_object' },
  }

  const resp = await fetch(`${process.env.MODEL_BASE_URL}/chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${process.env.MODEL_API_KEY}`,
    },
    body: JSON.stringify(body),
  })
  if (!resp.ok) throw new Error(`model error ${resp.status}: ${await resp.text()}`)
  const data = await resp.json()
  const text = data.choices?.[0]?.message?.content ?? '{}'
  return JSON.parse(text) as CocreateResponse
}
```

// src/sketch/renderSketch.ts
import type { SketchShapeSpec } from '../types/cocreate'

type Canvas = { width: number; height: number }
const GEO: Record<string, string> = { ellipse: 'ellipse', rectangle: 'rectangle' }

export function toTldrawShape(spec: SketchShapeSpec, canvas: Canvas) {
  const x = spec.x * canvas.width
  const y = spec.y * canvas.height
  const w = (spec.w ?? 0.1) * canvas.width
  const h = (spec.h ?? 0.1) * canvas.height
  return {
    id: `shape:${spec.id}`,
    type: 'geo',
    x,
    y,
    props: {
      geo: GEO[spec.kind] ?? 'ellipse',
      w,
      h,
      color: spec.color ?? 'black',
      dash: 'draw', // 手绘风
      fill: 'none',
    },
  }
}
```

/** 按 at 排序逐拍用 tldraw 动画把 shape 移到目标位姿 */
export async function runAnimation(editor: Editor, steps: AnimationStep[], canvas: { width: number; height: number }) {
  const ordered = [...steps].sort((a, b) => a.at - b.at)
  for (const step of ordered) {
    const id = `shape:${step.shapeId}` as never
    const shape = editor.getShape(id)
    if (!shape) continue
    editor.animateShape(
      {
        id,
        type: shape.type,
        x: step.x != null ? step.x * canvas.width : shape.x,
        y: step.y != null ? step.y * canvas.height : shape.y,
        rotation: step.rotation ?? shape.rotation,
        opacity: step.opacity ?? shape.opacity,
      } as never,
      { animation: { duration: step.durationMs ?? 600 } }
    )
    await new Promise((r) => setTimeout(r, step.durationMs ?? 600))
  }
}
```

> 注：`animateShape` 的精确签名以当前 tldraw 版本为准，执行时对照 `tldraw.dev/sdk-features/animation` 校验字段。

```typescript
// src/voice/useVoice.ts
import { useCallback, useRef, useState } from 'react'

export function useVoice() {
  const [recording, setRecording] = useState(false)
  const chunks = useRef<Blob[]>([])
  const recorder = useRef<MediaRecorder | null>(null)

  const start = useCallback(async () => {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    const mr = new MediaRecorder(stream)
    chunks.current = []
    mr.ondataavailable = (e) => chunks.current.push(e.data)
    mr.start()
    recorder.current = mr
    setRecording(true)
  }, [])

  const stop = useCallback(async (): Promise<string> => {
    const mr = recorder.current
    if (!mr) return ''
    const done = new Promise<Blob>((resolve) => {
      mr.onstop = () => resolve(new Blob(chunks.current, { type: 'audio/webm' }))
    })
    mr.stop()
    setRecording(false)
    const blob = await done
    const resp = await fetch('/api/stt', {
      method: 'POST',
      headers: { 'Content-Type': 'application/octet-stream' },
      body: await blob.arrayBuffer(),
    })
    const { text } = await resp.json()
    return text ?? ''
  }, [])

  const speak = useCallback(async (text: string) => {
    if (!text) return
    const resp = await fetch('/api/tts', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text }),
    })
    const buf = await resp.arrayBuffer()
    if (buf.byteLength === 0) return // TTS 未接入时跳过
    const url = URL.createObjectURL(new Blob([buf], { type: 'audio/mpeg' }))
    await new Audio(url).play()
  }, [])

  return { recording, start, stop, speak }
}
```

```tsx
// src/App.tsx
import { useRef, useState } from 'react'
import type { Editor } from 'tldraw'
import './App.css'
import { SketchCanvas } from './sketch/SketchCanvas'
import { toTldrawShape } from './sketch/renderSketch'
import { runAnimation } from './animation/runAnimation'
import { useVoice } from './voice/useVoice'
import { requestCocreate } from './cocreate/cocreateClient'
import { buildRequest } from './cocreate/buildRequest'
import { applyResponse, emptySession } from './store/sessionStore'
import type { SessionState } from './types/cocreate'

export default function App() {
  const editorRef = useRef<Editor | null>(null)
  const [session, setSession] = useState<SessionState>(emptySession())
  const [busy, setBusy] = useState(false)
  const [typed, setTyped] = useState('')
  const { recording, start, stop, speak } = useVoice()

  async function exportCanvasPng(): Promise<string> {
    const editor = editorRef.current
    if (!editor) return ''
    const ids = editor.getCurrentPageShapeIds()
    const { blob } = await editor.toImage([...ids], { format: 'png', background: true })
    return await new Promise((resolve) => {
      const reader = new FileReader()
      reader.onload = () => resolve(reader.result as string)
      reader.readAsDataURL(blob)
    })
  }

  async function sendTurn(utterance: string) {
    const editor = editorRef.current
    if (!editor || busy) return
    setBusy(true)
    try {
      const imageDataUrl = await exportCanvasPng()
      const payload = buildRequest({ imageDataUrl, childUtterance: utterance, history: session.dialog })
      const res = await requestCocreate(payload)

      // 新增 AI 简笔画
      const bounds = editor.getViewportPageBounds()
      const canvas = { width: bounds.w, height: bounds.h }
      for (const spec of res.addShapes ?? []) {
        editor.createShape(toTldrawShape(spec, canvas) as never)
      }
      if (res.animation?.length) await runAnimation(editor, res.animation, canvas)

      setSession((s) => applyResponse(s, res, utterance))
      await speak(res.speak || res.askChild || '')
    } finally {
      setBusy(false)
    }
  }

  return (
    <div className="shenbi">
      <SketchCanvas onMount={(e) => (editorRef.current = e)} />

      <div className="story-strip">
        {session.story.map((e) => (
          <p key={e.id}>{e.text}</p>
        ))}
      </div>

      <div className="talk-bar">
        <button
          className="talk-btn"
          onMouseDown={start}
          onMouseUp={async () => {
            const text = await stop()
            if (text) void sendTurn(text)
          }}
        >
          {recording ? '🎤 在听…' : '按住说话'}
        </button>
        <input
          value={typed}
          placeholder="（或在这里打字告诉我）"
          onChange={(e) => setTyped(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === 'Enter' && typed.trim()) {
              void sendTurn(typed.trim())
              setTyped('')
            }
          }}
        />
        {busy && <span className="thinking">✏️ 我在想…</span>}
      </div>
    </div>
  )
}
```
